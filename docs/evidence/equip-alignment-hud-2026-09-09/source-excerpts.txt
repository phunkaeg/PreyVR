32: struct CalibrationState {
33:     unsigned int pending = 0;        // asked for by ik.calibrate
34:     unsigned int autoPending = 0;    // re-queued by a re-bind, after settling
35:     unsigned int settleFrames = 0;
36:     unsigned int calibrated = 0;
37:     Quaternion offsets[2]{};
38:     std::uint64_t ownerGeneration = 0, referenceGeneration = 0, trackingEpoch = 0;
39:     // Re-binding after a weapon change, a recentre or a focus loss invalidates
40:     // the rotation offset -- it maps the controller onto *this* weapon's
41:     // authored grip, and the next weapon's grip is different.
42:     //
43:     // **Re-take it rather than merely dropping it.** Dropping alone leaves the
44:     // hand tracking position while its rotation silently stops, which a wearer
45:     // reported on 2026-09-08 after swapping weapons: "it seemed to reset the
46:     // rotational tracking and it was only tracking positionally". A hand that
47:     // half-works reads as a bug, not as a prompt to recalibrate, and asking a
48:     // player to recalibrate after every weapon swap is not a shippable answer.
49:     //
50:     // Two properties are preserved from the original contract:
51:     //
52:     //  * **No stale offset carries over** -- `calibrated` is still cleared.
53:     //  * **No stale *request* carries over** -- an outstanding request that
54:     //    never completed belonged to the old rig and is discarded. Only hands
55:     //    with a COMPLETED calibration are re-queued.
56:     //
57:     // And the re-take waits: `kSettleFrames` matched frames must pass first, so
58:     // the capture lands on the settled grip rather than midway through the
59:     // equip animation, which would bake a transient wrist pose into the offset.
60:     // A manual `ik.calibrate` is deliberate and commits on the next frame.
61:     static constexpr unsigned int kSettleFrames = 90;
62: 
63:     bool Bind(std::uint64_t owner, std::uint64_t reference, std::uint64_t epoch) {
64:         if (ownerGeneration == owner && referenceGeneration == reference && trackingEpoch == epoch) { return false; }
65:         pending = 0;                 // whatever the old rig had outstanding
66:         // A second rebind may arrive before the first automatic retake. In
67:         // that interval calibrated is zero, but autoPending still records a
68:         // hand which completed calibration before the transition. Preserve
69:         // that recovery intent across churn; never carry its old offset as valid.
70:         autoPending = (ownerGeneration != 0) ? (calibrated | autoPending) : 0u;
71:         settleFrames = 0;
72:         calibrated = 0;
73:         ownerGeneration = owner; referenceGeneration = reference; trackingEpoch = epoch;
74:         return true;
75:     }
76: 
77:     // One matched frame on the owning rig.
78:     void Tick() { if (settleFrames < kSettleFrames) { ++settleFrames; } }
79: 
80:     void Request(unsigned int hands) { pending |= hands & 3u; }
81:     // Explicit rig-signature reset is different from Bind's automatic recovery.
82:     void Invalidate() { pending = autoPending = settleFrames = calibrated = 0; }
83:     bool Pending(unsigned int hand) const {
84:         if (hand >= 2) { return false; }
85:         if (pending & (1u << hand)) { return true; }   // asked for deliberately
86:         return (autoPending & (1u << hand)) && settleFrames >= kSettleFrames;
87:     }
88:     // Queued by a re-bind but still settling. Reported so a wearer whose
89:     // rotation has not resumed yet can see that it is coming, not broken.
90:     bool Settling(unsigned int hand) const {
91:         return hand < 2 && (autoPending & (1u << hand)) && settleFrames < kSettleFrames;
92:     }
93:     void Commit(unsigned int hand, const Quaternion& offset) {
94:         if (hand >= 2) { return; }
95:         offsets[hand] = offset;
96:         calibrated |= 1u << hand;
97:         pending &= ~(1u << hand);
98:         autoPending &= ~(1u << hand);
99:     }
100: };
