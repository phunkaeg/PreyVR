   26: // The upper body for a bend `b` (radians; its direction is where the top goes)
   27: // on top of `base`, the hips moved by `shift` after it. Fills hip, spine, neck

   30: {
   31:     plan.hip = Then(base, Shift(shift));
   32:     const float theta = std::sqrt(b.x * b.x + b.y * b.y);
   33:     // Turning about (-b.y, b.x, 0) by +theta tips +Z towards b.
   34:     const Vec3 axis = theta > 1e-6f ? Vec3{-b.y / theta, b.x / theta, 0.0f} : Vec3{1, 0, 0};

   37:         const Vec3 pivot = Apply(t, s.spine[k]);

  100: void FootPlanter::Update(const Vec3 natural[2], const Vec3 thigh[2], const float leg[2], float bodySpeed, float dt,
  101:                          bool free, const FootSettings& s, Vec3 out[2])
  102: {
  103:     dt = std::isfinite(dt) ? std::clamp(dt, 0.0f, 0.1f) : 0.0f;
  104:     bool far = false;
  105:     for (int k = 0; k < 2; ++k) { far = far || !Finite(natural[k]) || Length(Sub(planted_[k], natural[k])) > 1.0f; }
  106:     if (free || !valid_ || far) {
  107:         for (int k = 0; k < 2; ++k) { planted_[k] = out[k] = natural[k]; stepping_[k] = false; }
  108:         valid_ = true;
  109:         return;
  110:     }
  111:     // A step lands where the body would stand the foot now.
  112:     for (int k = 0; k < 2; ++k) {
  113:         if (!stepping_[k]) { continue; }
  114:         t_[k] += dt / std::max(duration_[k], 1e-3f);
  115:         if (t_[k] >= 1.0f) { stepping_[k] = false; planted_[k] = natural[k]; ++steps_; }
  116:     }
  117:     // The next step: the foot most in need, one at a time unless a leg cannot reach.
  118:     float need[2]{};
  119:     bool reachless[2]{};
  120:     for (int k = 0; k < 2; ++k) {
  121:         const Vec3 d = Sub(planted_[k], natural[k]);
  122:         need[k] = std::sqrt(d.x * d.x + d.y * d.y) / std::max(s.drift, 1e-3f);
  123:         // Past what the leg can reach -- or past what the animation's own stance
  124:         // asks of it, which may already hold a leg nearly straight.
  125:         const float limit = std::max(s.reach * leg[k], Length(Sub(natural[k], thigh[k])) + 0.03f);
  126:         reachless[k] = Length(Sub(planted_[k], thigh[k])) > limit;
  127:     }
  128:     for (int k = 0; k < 2; ++k) {
  129:         const int other = 1 - k;
  130:         const bool wants = need[k] > 1.0f || reachless[k];
  131:         const bool first = need[k] >= need[other] || !(need[other] > 1.0f || reachless[other]);
  132:         if (!stepping_[k] && wants && (reachless[k] || (!stepping_[other] && first))) {
  133:             stepping_[k] = true;
  134:             from_[k] = planted_[k];
  135:             t_[k] = 0.0f;
  136:             duration_[k] = std::clamp(s.duration - 0.1f * std::max(bodySpeed, 0.0f), s.minDuration, s.duration);
  137:         }
  138:     }
  139:     for (int k = 0; k < 2; ++k) {
  140:         if (stepping_[k]) {
  141:             const float t = std::clamp(t_[k], 0.0f, 1.0f);
  142:             const float e = t * t * (3.0f - 2.0f * t);
  143:             out[k] = Add(from_[k], Mul(Sub(natural[k], from_[k]), e));
  144:             out[k].z += s.lift * std::sin(3.14159265f * t);
  145:         } else {
  146:             // Planted where it stands; its height follows the floor the body stands on.
  147:             out[k] = {planted_[k].x, planted_[k].y, natural[k].z};
  148:         }
  149:     }

  157: Plan Solve(const Skeleton& s, const Goal& goal, const Limits& limits)
  158: {
  159:     Plan plan{};
  160:     const float w = std::isfinite(goal.weight) ? std::clamp(goal.weight, 0.0f, 1.0f) : 0.0f;
  161:     if (!Finite(s.head) || !Finite(s.neck) || !Finite(s.hip) || !Finite(goal.head) || !std::isfinite(goal.yaw)) {
  162:         return plan;
  163:     }
  164:     const Vec3 target = Add(s.head, Mul(Sub(goal.head, s.head), w));
  165:     if (Length(Sub(target, s.head)) > 1.5f) { return plan; }   // not a body any more: leave it to the game
  166: 
  167:     // The turn, about the vertical through the head's pivot.
  168:     const Quaternion yawQ = AxisAngle({0, 0, 1}, goal.yaw * w);
  169:     const Rigid turn = TorsoTurn(s, goal);
  170:     const Vec3 forward = Rotate(yawQ, {0, 1, 0});
  171:     const Vec3 right{forward.y, -forward.x, 0.0f};
  172:     // A squat sits the hips back.
  173:     const float drop = std::max(0.0f, s.head.z - target.z);
  174:     const Rigid base = Then(turn, Shift(Mul(forward, -std::min(limits.maxSitBack, limits.sitBack * drop))));
  175: 
  176:     // The bend that puts the head's pivot over the goal, within the spine's range.
  177:     const float lever = std::max(0.2f, s.head.z - s.spine[0].z);
  178:     Vec3 b{};
  179:     Vec3 reached = Chain(s, base, b, {}, limits, plan);
  180:     for (int i = 0; i < 8; ++i) {
  181:         const Vec3 e{target.x - reached.x, target.y - reached.y, 0.0f};
  182:         if (Length(e) < 1e-4f) { break; }
  183:         b = Add(b, Mul(e, 1.0f / lever));
  184:         const float bf = Dot(b, forward), bs = Dot(b, right);
  185:         const float lf = bf >= 0.0f ? limits.forward : limits.back;
  186:         const float r = (bf / lf) * (bf / lf) + (bs / limits.side) * (bs / limits.side);
  187:         if (r > 1.0f) { b = Mul(b, 1.0f / std::sqrt(r)); }
  188:         reached = Chain(s, base, b, {}, limits, plan);
  189:     }
  190:     // What the spine could not reach, the hips slide a little towards.
  191:     Vec3 shift{target.x - reached.x, target.y - reached.y, 0.0f};
  192:     const float rest = Length(shift);
  193:     if (rest > limits.slide) { shift = Mul(shift, limits.slide / rest); }
  194: 
  195:     // The height: the hips drop or rise the rest, as far as the legs allow.
  196:     float dz = std::max(target.z - reached.z, -limits.maxDrop);
  197:     if (s.legs) {
  198:         for (int k = 0; k < 2; ++k) {
  199:             const Vec3 a0 = Add(Apply(base, s.thigh[k]), shift);
  200:             const Vec3 foot = goal.footSet[k] ? goal.foot[k] : Apply(turn, s.foot[k]);
  201:             const float full = (Length(Sub(s.knee[k], s.thigh[k])) + Length(Sub(s.foot[k], s.knee[k]))) *
  202:                                limits.straight;
  203:             const float hx = a0.x - foot.x, hy = a0.y - foot.y;
  204:             const float h2 = hx * hx + hy * hy;
  205:             const float up = full * full > h2 ? foot.z + std::sqrt(full * full - h2) - a0.z : 0.0f;
  206:             dz = std::min(dz, up);
  207:         }
  208:     }
  209:     shift.z = dz;
  210:     reached = Chain(s, base, b, shift, limits, plan);
  211:     if (s.legs) {
  212:         for (int k = 0; k < 2; ++k) { Leg(s, k, turn, plan.hip, forward, goal, plan); }
  213:     } else {
  214:         for (int k = 0; k < 2; ++k) { plan.thigh[k] = plan.shin[k] = plan.hip; plan.foot[k] = plan.hip; }
  215:     }
  216:     plan.bend = std::sqrt(b.x * b.x + b.y * b.y);
  217:     plan.hipShift = shift;
  218:     plan.headReached = reached;
  219:     plan.miss = Length(Sub(target, reached));
  220:     plan.valid = Finite(reached);
  221:     return plan;
  222: }
