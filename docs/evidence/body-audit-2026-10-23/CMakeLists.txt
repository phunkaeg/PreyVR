cmake_minimum_required(VERSION 3.24)
project(PreyVRBodyAudit LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT REVIEW_SOURCE OR NOT REVIEW_BUILD OR NOT MINHOOK_SOURCE OR NOT MINHOOK_LIB)
  message(FATAL_ERROR "Pass the reviewed source/build and pinned MinHook paths")
endif()
add_library(review_core STATIC IMPORTED)
set_target_properties(review_core PROPERTIES IMPORTED_LOCATION "${REVIEW_BUILD}/Release/preyvr_core.lib")
foreach(probe crouch geometry torso)
  add_executable(body_${probe}_probe ${probe}.cpp)
  target_include_directories(body_${probe}_probe PRIVATE "${REVIEW_SOURCE}/include" "${REVIEW_SOURCE}/src/dll" "${REVIEW_SOURCE}/src/lifecycle" "${MINHOOK_SOURCE}/include")
  target_compile_definitions(body_${probe}_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(body_${probe}_probe PRIVATE /Gy)
  target_link_options(body_${probe}_probe PRIVATE /OPT:REF)
  target_link_libraries(body_${probe}_probe PRIVATE review_core "${MINHOOK_LIB}" psapi)
endforeach()
